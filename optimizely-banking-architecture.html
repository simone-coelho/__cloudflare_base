Optimizely Mobile Banking Personalization Architecture

Dynamic Content Personalization Without App Redeployment

Executive Summary

This architecture enables your banking mobile applications to deliver personalized experiences to customers based on their segments and behaviors, all without requiring app redeployment. We present two implementation approaches:

Use Case 1: Direct Variable-Driven Personalization - Content delivered through Optimizely Feature Variables
Use Case 2: CMS/GraphQL-Driven Personalization - Content retrieved from Optimizely CMS via GraphQL

Both approaches integrate with Microsoft Dynamics CDP for real-time customer segmentation.

Use Case 1: Direct Variable-Driven Personalization

Architecture Overview

Sequence Diagram

Variable to UI Mapping - Exploded View

How it works:
  1. JSON Variables are defined in Optimizely for each targeted delivery variation
  2. Values are extracted from the JSON and mapped to specific UI component properties
  3. Components are created dynamically using these values (images, text, colors, actions)
  4. UI is assembled at runtime into a complete carousel experience
  5. No app deployment needed - just update the JSON values in Optimizely!

// CDPSegmentBridge.swift
class CDPSegmentBridge {
    private let cdpAPIClient: DynamicsCDPClient
    private let cache: SegmentCache
    
    func fetchSegmentsForUser(userId: String) async throws -> [String: Any] {
        // Check cache first
        if let cachedSegments = cache.getSegments(for: userId) {
            return cachedSegments
        }
        
        // Call Microsoft Dynamics CDP API
        let cdpResponse = try await cdpAPIClient.getCustomerSegments(
            customerId: userId,
            apiKey: Configuration.cdpAPIKey
        )
        
        // Transform CDP response to Optimizely-compatible format
        let optimizelyAttributes = [
            "segments": cdpResponse.segments,
            "customer_tier": cdpResponse.customerTier,
            "lifetime_value": cdpResponse.ltv,
            "is_mortgage_interested": cdpResponse.segments.contains("mortgage_browsers"),
            "is_credit_card_prospect": cdpResponse.segments.contains("cc_applicants")
        ]
        
        // Cache for 5 minutes
        cache.store(segments: optimizelyAttributes, for: userId, ttl: 300)
        
        return optimizelyAttributes
    }
}

// PersonalizedCarouselBuilder.swift
class PersonalizedCarouselBuilder {
    private let optimizelyClient: OptimizelyClient
    private let cdpBridge: CDPSegmentBridge
    
    func buildCarousel(for userId: String) async -> CarouselView {
        do {
            // 1. Fetch CDP segments
            let segments = try await cdpBridge.fetchSegmentsForUser(userId: userId)
            
            // 2. Create user context with segments
            let userContext = optimizelyClient.createUserContext(
                userId: userId,
                attributes: segments
            )
            
            // 3. Get decision with variation
            let decision = userContext.decide(key: "banking_carousel_personalization")
            
            // 4. Extract the JSON variable
            guard let carouselConfig = decision.variables.toJSON()["carousel_config"] as? [String: Any],
                  let cardsData = carouselConfig["cards"] as? [[String: Any]] else {
                return createDefaultCarousel()
            }
            
            // 5. Build cards from variable values
            let cards = cardsData.compactMap { cardData -> CardView? in
                guard let imageUrl = cardData["imageUrl"] as? String,
                      let headline = cardData["headline"] as? String,
                      let ctaText = cardData["ctaText"] as? String else {
                    return nil
                }
                
                return CardView(
                    imageUrl: URL(string: imageUrl),
                    headline: headline,
                    ctaText: ctaText,
                    backgroundColor: Color(hex: cardData["backgroundColor"] as? String ?? "#000"),
                    action: {
                        self.handleAction(cardData["ctaAction"] as? String)
                    }
                )
            }
            
            // 6. Create carousel with configuration
            return CarouselView(
                cards: cards,
                maxVisible: carouselConfig["maxCards"] as? Int ?? 3,
                autoRotate: carouselConfig["autoRotate"] as? Bool ?? false
            )
            
        } catch {
            print("Error building personalized carousel: \(error)")
            return createDefaultCarousel()
        }
    }
    
    private func handleAction(_ action: String?) {
        guard let action = action else { return }
        
        switch action {
        case "openMortgageCalc":
            NavigationManager.shared.navigate(to: .mortgageCalculator)
        case "startMortgageApp":
            NavigationManager.shared.navigate(to: .mortgageApplication)
        case "viewCreditCards":
            NavigationManager.shared.navigate(to: .creditCardOffers)
        default:
            break
        }
    }
}

// Usage in View Controller
class HomeViewController: UIViewController {
    private let carouselBuilder = PersonalizedCarouselBuilder()
    private var carouselView: CarouselView?
    
    override func viewDidLoad() {
        super.viewDidLoad()
        loadPersonalizedContent()
    }
    
    private func loadPersonalizedContent() {
        Task {
            let userId = UserManager.shared.currentUserId
            let carousel = await carouselBuilder.buildCarousel(for: userId)
            
            await MainActor.run {
                self.carouselView = carousel
                self.view.addSubview(carousel)
                // Layout constraints...
            }
        }
    }
}

// ContentDrivenCarouselBuilder.swift
class ContentDrivenCarouselBuilder {
    private let optimizelyClient: OptimizelyClient
    private let cdpBridge: CDPSegmentBridge
    private let graphQLClient: OptimizelyGraphClient
    
    func buildCarousel(for userId: String) async -> CarouselView {
        do {
            // 1. Fetch CDP segments
            let segments = try await cdpBridge.fetchSegmentsForUser(userId: userId)
            
            // 2. Create user context
            let userContext = optimizelyClient.createUserContext(
                userId: userId,
                attributes: segments
            )
            
            // 3. Get decision - returns content IDs instead of full content
            let decision = userContext.decide(key: "banking_carousel_cms")
            
            // 4. Extract content IDs from decision variables
            guard let contentConfig = decision.variables.toJSON()["content_config"] as? [String: Any],
                  let contentIds = contentConfig["card_ids"] as? [String] else {
                return createDefaultCarousel()
            }
            
            // 5. Build GraphQL query for content
            let query = buildContentQuery(contentIds: contentIds)
            
            // 6. Fetch content from CMS via GraphQL
            let contentResponse = try await graphQLClient.query(query)
            
            // 7. Build carousel from CMS content
            return buildCarouselFromContent(contentResponse)
            
        } catch {
            print("Error building content-driven carousel: \(error)")
            return createDefaultCarousel()
        }
    }
    
    private func buildContentQuery(contentIds: [String]) -> String {
        return """
        query GetCarouselContent {
            CardContent(
                where: { ContentLink: { Id: { in: [\(contentIds.joined(separator: ","))] } } }
            ) {
                items {
                    contentLink {
                        id
                    }
                    headline
                    bodyText
                    ctaText
                    ctaAction
                    backgroundHexColor
                    heroImage {
                        url
                        alt
                    }
                    targeting {
                        segment
                        priority
                    }
                }
            }
        }
        """
    }
    
    private func buildCarouselFromContent(_ response: GraphQLResponse) -> CarouselView {
        guard let items = response.data?.cardContent?.items else {
            return createDefaultCarousel()
        }
        
        let cards = items.compactMap { item -> CardView? in
            guard let imageUrl = item.heroImage?.url,
                  let headline = item.headline,
                  let ctaText = item.ctaText else {
                return nil
            }
            
            return CardView(
                imageUrl: URL(string: imageUrl),
                headline: headline,
                bodyText: item.bodyText,
                ctaText: ctaText,
                backgroundColor: Color(hex: item.backgroundHexColor ?? "#000"),
                action: {
                    self.handleAction(item.ctaAction)
                }
            )
        }
        
        return CarouselView(cards: cards)
    }
}

// GraphQL Client
class OptimizelyGraphClient {
    private let endpoint = "https://graph.optimizely.com/content/v2"
    private let apiKey: String
    
    func query(_ graphQLQuery: String) async throws -> GraphQLResponse {
        var request = URLRequest(url: URL(string: endpoint)!)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue("Bearer \(apiKey)", forHTTPHeaderField: "Authorization")
        
        let body = ["query": graphQLQuery]
        request.httpBody = try JSONSerialization.data(withJSONObject: body)
        
        let (data, _) = try await URLSession.shared.data(for: request)
        return try JSONDecoder().decode(GraphQLResponse.self, from: data)
    }
}

Comparison and Recommendations

Aspect Use Case 1: Direct Variables Use Case 2: CMS/GraphQL
Setup Complexity Simple - Only Optimizely setup Complex - Requires CMS + GraphQL
Content Management Through Optimizely UI only Full CMS capabilities
Performance Faster - Single API call Slower - Multiple API calls
Content Richness Limited to variable types Unlimited - full CMS features
Offline Support Better - Can cache decisions Challenging - Needs content sync
Team Workflow Marketers work in Optimizely Content team in CMS, Marketers in Optimizely
Best For Simple personalization, A/B testing Complex content, multiple teams

Our Recommendation

Start with Use Case 1 (Direct Variables) for immediate value and simpler implementation. This approach:

  • Gets you to market faster
  • Requires less infrastructure
  • Provides excellent performance
  • Enables powerful personalization

Migrate to Use Case 2 when you need:

  • Rich media content management
  • Multiple content editors
  • Complex content workflows
  • Integration with existing CMS content
